import random
import numpy as np
import pandas as pd
import tensorflow as tf
from tensorflow.keras.models import Sequential
from tensorflow.keras.layers import Dense
from tensorflow.keras.layers import LSTM, Input
from sklearn.preprocessing import MinMaxScaler
from sklearn.metrics import mean_squared_error
from tensorflow.python.client import device_lib
import time as tt
import itertools

import os
for dirname, _, filenames in os.walk('/kaggle/input'):
    for filename in filenames:
        print(os.path.join(dirname, filename))

#NOMENCLATURA
# _u --> unscaled (scaler.inverse_transform)
# _s --> static
# _o --> online learning
# nada --> scaled (scaler.transform)

# GET CPU AND GPU
def get_available_devices():
    local_device_protos = device_lib.list_local_devices()
    return [x.name for x in local_device_protos]
    
# FUNCION RMSE
def rmse_loss(y_true, y_pred):
    return tf.sqrt(tf.reduce_mean(tf.square(y_true - y_pred)))    
    
# ARRAY TO MATRIX
def create_dataset(dataset, look_back=1):
    dataX, dataY = [], []
    for i in range(len(dataset) - look_back - 1):
        a = dataset[i:(i + look_back), :]
        dataX.append(a)
        dataY.append(dataset[i + look_back, :])
    return np.array(dataX), np.array(dataY)

# Callback para resetear los estados de forma ultrarrapida
class ResetStatesCallback(tf.keras.callbacks.Callback):
    def on_epoch_end(self, epoch, logs=None):
        for layer in self.model.layers:
            if hasattr(layer, 'reset_states'):
                layer.reset_states()


# tf.function para invocar Tracing Graphs
@tf.function
def online_train_step(model, optimizer, x, y, norm_sq):
    with tf.GradientTape() as tape:
        predictions = model(x, training=True)
        loss = rmse_loss(y, predictions)
    
    gradients = tape.gradient(loss, model.trainable_variables)
    adjusted_gradients = [g * norm_sq for g in gradients]
    optimizer.apply_gradients(zip(adjusted_gradients, model.trainable_variables))

#Restricciones de reproducibilidad
os.environ['PYTHONHASHSEED'] = '0'
os.environ['TF_DETERMINISTIC_OPS'] = '1'
os.environ['TF_CUDNN_DETERMINISTIC'] = '1'
random.seed(7)
np.random.seed(7)
tf.random.set_seed(7)
tf.config.experimental.enable_op_determinism()

#Lectura del dataset
dataframe = pd.read_csv('/kaggle/input/datasets/username/traffic-7x7-1/Traffic_7x7_1.csv', usecols=range(1,50), engine='python')
dataset_full = dataframe.values
dataset_full = dataset_full.astype('float32')
num_features = dataset_full.shape[1] #7x7=49
total_rows = dataset_full.shape[0]

#GRID SEARCH EJEMPLO
hyperparameters = {
    'look_back': [16, 18],
    'batch_size': [22, 23, 24],
    'n_epochs_update': [6, 7],
    'threshold': [0.13, 0.15],
}

combinations = list(itertools.product(*hyperparameters.values()))
output_csv_file = '/kaggle/working/GridSearch_LSTM.csv'
grid_search_summary = []


#BUCLE GRID SEARCH
for combi in combinations:
    #Borrar todo rastro de modelos de otras iteraciones para evitar influencias no autorizadas
    tf.keras.backend.clear_session()
    
    #Restricciones de reproducibilidad de nuevo, por si acaso
    os.environ['PYTHONHASHSEED'] = '0'
    os.environ['TF_DETERMINISTIC_OPS'] = '1'
    os.environ['TF_CUDNN_DETERMINISTIC'] = '1'
    random.seed(7)
    np.random.seed(7)
    tf.random.set_seed(7)
    tf.config.experimental.enable_op_determinism()
    
    #Combinacion de hiperparametros actual
    params = dict(zip(hyperparameters.keys(), combi))
    current_look_back = params['look_back']
    current_batch_size = params['batch_size']
    current_n_epochs_update = params['n_epochs_update']
    current_threshold = params['threshold']
    
    # Separacion entre train y test de forma automatica segun batch_size,
    # el numero total de muestras de los datasets ha de ser divisible por batch_size 
    desired_train_size = 400
    train_available_batches = (desired_train_size - current_look_back - 1) // current_batch_size
    train_size = current_batch_size * train_available_batches + current_look_back + 1
    #create_dataset se come las primeras look_back + 1 muestras para crear el dataset de test/train
    available_batches = (total_rows - current_look_back -1 - train_size) // current_batch_size
    test_size = current_batch_size * available_batches + current_look_back +1
    
    dataset = dataset_full[0:train_size+test_size]

    scaler = MinMaxScaler(feature_range=(0, 1))
    train = dataset[0:train_size, :]
    test = dataset[train_size : train_size+test_size, :]
    
    #EL MODELO ORIGINAL HACE FIT_TRANSFORM CON TODO EL DATASET --> DATA LEAKAGE
    train = scaler.fit_transform(train)
    test = scaler.transform(test)

    # X=[T,T+LOOKBACK]  Y=[T+LOOCKBACK+1]
    # look_back define cuantas muestras se utiliza para las series temporales
    look_back = current_look_back  #este parametro, permite definir una ventana deslizante
    trainX, trainY = create_dataset(train, look_back)
    testX, testY = create_dataset(test, look_back)

    #batch_size indica cuantas ventanas look_back se procesan antes de actualizar los pesos
    batch_size = current_batch_size
    # Cuantas vueltas al dataset entero se dan para entrenar
    n_epochs = 400

    # MODELO LSTM CON CAPA INPUT
    myLSTM = Sequential()
    myLSTM.add(Input(batch_shape=(batch_size, trainX.shape[1], trainX.shape[2])))
    myLSTM.add(LSTM(look_back, stateful=True))
    myLSTM.add(Dense(trainX.shape[2]))
    myLSTM.compile(loss='mean_squared_error', optimizer='adam')
    myLSTM.summary()

    #Entrenamiento optimizado ultrarrapido
    #El reseteo del estado debe hacerse asi cada epoch. Hacerlo es recomendable, si bien para datasets pequenos no es relevante
    print('Fitting myLSTM on verbose 0')
    myLSTM.fit(
        trainX, trainY,
        epochs=n_epochs,
        batch_size=batch_size,
        verbose=0,
        shuffle=True,
        callbacks=[ResetStatesCallback()]
    )
    #El myLSTMos LSTM es stateful, es decir, que guarda sus estados h(t) y c(t) y los usa como valor inicial en el siguiente batch
    #Como shuffle=True, el orden de los batches es aleatorio. Lo que significa que actua como un stateless.
    #Shuffle=True mejora la generalizacion.
    
    #TEST ESTATICO
    train_Y_u = scaler.inverse_transform(trainY)
    test_Y_u = scaler.inverse_transform(testY)

    train_Yhat_s = myLSTM.predict(trainX, batch_size=batch_size)
    train_Yhat_su = scaler.inverse_transform(train_Yhat_s)

    test_Yhat_s = myLSTM.predict(testX, batch_size=batch_size)
    test_Yhat_su = scaler.inverse_transform(test_Yhat_s)

    # ONLINE LEARNING TEST
    n_epochs_update = current_n_epochs_update
    error_threshold = current_threshold
    rmse_norm = 1
    online_yhat_u = list()
    times = list()
    optimizer = tf.keras.optimizers.Adam()
    optimizer.build(myLSTM.trainable_variables)

    # Inicializar el modelo con un primer batch para que los estados h(t) y c(t) no esten vacios.
    online_X = trainX[-batch_size:,:,:]
    online_Y = trainY[-batch_size:,:]
    
    for i in range(testY.shape[0]-batch_size):
        if rmse_norm > error_threshold:
            #print(f"RMSE NORM for sample {i}: {rmse_norm}")
            #Temporizador para online learning
            st = tt.time()
            # Conversion a tf tensor para el Tracing Graph
            norm_sq_tensor = tf.cast(rmse_norm**2, tf.float32) 
        
            for j in range(n_epochs_update):
            # Llamada al Tracing Graph ultrarrapido para online learning
                online_train_step(myLSTM, optimizer, online_X, online_Y, norm_sq_tensor) 
            
            # Resetear estados
            for layer in myLSTM.layers:
                if hasattr(layer, 'reset_states'):
                    layer.reset_states()
            
            #Parar temporizador
            et = tt.time()
            elapsed = et-st
            times.append(elapsed)
            
            
        # Siguiente batch en orden cronologico
        newX, newY = testX[i:i+batch_size,:,:], testY[i:i+batch_size,:]
        
        # Predicciones con el batch siguiente
        online_yhat = myLSTM.predict(newX,batch_size=batch_size)
        online_yhat = scaler.inverse_transform(online_yhat)
        newY_u = scaler.inverse_transform(newY)
        
        # NRMSE de la prediccion
        rmse_norm =((np.sqrt(np.mean(((newY_u[-1,:] - online_yhat[-1,:])/online_yhat[-1,:])**2))))
        
        online_yhat_u.append(online_yhat[-1])
        
        # Mover la ventana del batch
        online_X = newX #Meto solo un batch en el nuevo entrenamiento (no lo meneo mucho)
        online_Y = newY
        #Empezar de nuevo con el nuevo batch (procesar n_epoch_updates veces, coger el batch siguiente, predecir sobre ese batch)

    test_Yhat_ou=np.array(online_yhat_u)
    times=np.array(times)

    #Calcular NRMSE para todas las celdas
    trainScore=np.empty(num_features)
    testScore=np.empty(num_features)
    testScore_inc=np.empty(num_features)

    for i in range(num_features):
        trainScore[i] = np.sqrt(mean_squared_error(train_Y_u[:,i], train_Yhat_su[:,i]))/np.std(train_Y_u[:,i])
        testScore[i] = np.sqrt(mean_squared_error(test_Y_u[:,i], test_Yhat_su[:,i]))/np.std(test_Y_u[:,i])
        testScore_inc[i] = np.sqrt(mean_squared_error(test_Y_u[batch_size-1:-1,i], test_Yhat_ou[:,i]))/np.std(test_Y_u[batch_size-1:-1,i])

    timeScore = np.mean(times)
    avg_train_score = np.mean(trainScore)
    avg_test_score = np.mean(testScore)
    avg_test_score_inc = np.mean(testScore_inc)
    
    grid_search_summary = {
        'lookback': current_look_back,
        'batchsize': current_batch_size,
        'updates': current_n_epochs_update,
        'threshold': current_threshold,
        'AvgTrain_RNMSE': avg_train_score,
        'AvgTestRNMSE': avg_test_score,
        'AvgTestIncRNMSE': avg_test_score_inc,
        'AvgTestIncTime': timeScore
    }
    
    combi_df = pd.DataFrame([grid_search_summary])
    
    #Crear csv con los resultados
    file_exists = os.path.isfile(output_csv_file)
    combi_df.to_csv(output_csv_file, mode='a', header=not file_exists, index=False)
